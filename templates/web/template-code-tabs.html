    💻 Code Template with Tabs

    This template demonstrates the code component with tabs, which allows showing different versions of the same code (for example, Java 7, Java 8, Java 9, etc.) in an organized and easy-to-navigate way.

List names = Arrays.asList("Java", "Python", "C++");
Collections.sort(names, new Comparator() {
    @Override
    public int compare(String s1, String s2) {
        return s1.compareTo(s2);
    }

    JAVA 8 
    // Java 8 - Using Lambda Expressions
List names = Arrays.asList("Java", "Python", "C++");
Collections.sort(names, (s1, s2) -> s1.compareTo(s2));

// Or using method reference
names.sort(String::compareTo);

// Using Streams
names.stream()
    .sorted()
    .forEach(System.out::println);
    JAVA 9 
    // Java 9 - List.of() to create immutable lists
List names = List.of("Java", "Python", "C++");

// Improved Streams
names.stream()
    .sorted()
    .forEach(System.out::println);

// Optional.ifPresentOrElse
Optional first = names.stream().findFirst();
first.ifPresentOrElse(
    System.out::println,
    () -> System.out.println("Empty list")
);
    JAVA 11 
    // Java 11 - Improved String methods
List names = List.of("Java", "Python", "C++");

// String.isBlank(), String.lines()
String text = "Java\nPython\nC++";
text.lines()
    .filter(line -> !line.isBlank())
    .forEach(System.out::println);

// var in lambdas
var result = names.stream()
    .map(String::toUpperCase)
    .toList();

    📝 Required JavaScript

    JAVASCRIPT 
    function switchCodeTab(button, tabId) {
    // Remove active from all buttons and contents
    const container = button.closest('.code-tabs-container');
    container.querySelectorAll('.code-tab-button').forEach(btn => {
        btn.classList.remove('active');
    });
    container.querySelectorAll('.code-tab-content').forEach(content => {
        content.classList.remove('active');
    });
    
    // Add active to clicked button and corresponding content
    button.classList.add('active');
    const content = document.getElementById(tabId);
    if (content) {
        content.classList.add('active');
        
        // Re-apply highlight.js if necessary
        if (typeof hljs !== 'undefined') {
            content.querySelectorAll('pre code').forEach((block) => {
                hljs.highlightElement(block);
                if (typeof hljs.lineNumbersBlock !== 'undefined') {
                    hljs.lineNumbersBlock(block);
                }
            });
        }
    }
}

function copyCode(button) {
    const codeBlock = button.closest('.code-block-wrapper').querySelector('code');
    const text = codeBlock.textContent;
    navigator.clipboard.writeText(text).then(() => {
        button.textContent = 'Copied!';
        button.classList.add('copied');
        setTimeout(() => {
            button.textContent = 'Copy';
            button.classList.remove('copied');
        }, 2000);
    });
}
    💡 Tip: The code component with tabs is ideal for showing code evolution, different implementations, or comparisons between versions/frameworks. Each tab can have its own code with independent syntax highlighting.
    ⚠️ Important: Make sure to include the JavaScript function switchCodeTab() on the page so the tabs work correctly. The function also re-applies syntax highlighting when a tab is selected.
